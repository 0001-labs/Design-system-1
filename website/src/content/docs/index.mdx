```html
<!-- Classless: native HTML, no JavaScript -->
<link
  rel="stylesheet"
  href="https://cdn.jsdelivr.net/npm/ds-one@alpha/DS1/1-root/one.semantic.css"
/>

<!-- Components: tokens plus the Web Component bundle -->
<link
  rel="stylesheet"
  href="https://cdn.jsdelivr.net/npm/ds-one@alpha/DS1/1-root/one.css"
/>
<script
  type="module"
  src="https://cdn.jsdelivr.net/npm/ds-one@alpha/dist/ds-one.bundle.min.js"
></script>
```

```bash
bun add ds-one@alpha
```

```typescript
import "ds-one";
import "ds-one/styles";
```
